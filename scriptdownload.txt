    function showToast(msg) {
      const toast = document.getElementById('toast');
      toast.textContent = msg;
      toast.style.display = 'block';
      setTimeout(() => { toast.style.display = 'none'; }, 3500);
    }

    function getProgress(id) {
      try {
        const raw = localStorage.getItem('unlock_progress_' + id);
        return raw ? JSON.parse(raw) : { count: 0, unlocked: false };
      } catch (e) {
        return { count: 0, unlocked: false };
      }
    }

    function saveProgress(id, count, unlocked) {
      localStorage.setItem('unlock_progress_' + id, JSON.stringify({ count, unlocked }));
      renderFiles();
    }

    // THE USER'S MONETAG REWARDED POPUP CALL
    function watchRewardedAd(fileId) {
      const file = files.find(f => f.id === fileId);
      if (!file) return;

      if (typeof window.show_11999946 === 'function') {
        showToast('বিজ্ঞাপন লোড হচ্ছে...');
        // Execute Monetag show_11999946('pop')
        window.show_11999946('pop').then(() => {
          // user watch ad till the end or close it in interstitial format
          // your code to reward user for rewarded format:
          addReward(fileId);
        }).catch(e => {
          // user get error during playing ad
          console.warn('Monetag error or ad closed early', e);
          // Fallback simulation so user is not stuck:
          setTimeout(() => addReward(fileId), 3000);
          showToast('বিজ্ঞাপন দেখতে কিছুটা সময় লাগছে, ৩ সেকেন্ড অপেক্ষা করুন...');

        setTimeout(() => addReward(fileId), 3000);
      }
    }

    function addReward(fileId) {
      const file = files.find(f => f.id === fileId);
      const cur = getProgress(fileId);
      const newCount = cur.count + 1;
      const unlocked = newCount >= file.requiredAds;
      saveProgress(fileId, newCount, unlocked);

      if (unlocked && !cur.unlocked) {
        showToast('🎉 অভিনন্দন! ' + file.title + ' আনলক হয়েছে!');
      } else {
        showToast('+১ টি বিজ্ঞাপন সম্পন্ন (' + newCount + '/' + file.requiredAds + ')');
      }
    }

    function renderFiles(filterText = '') {
      const grid = document.getElementById('filesGrid');
      grid.innerHTML = '';

      const filtered = files.filter(f => 
        f.title.toLowerCase().includes(filterText.toLowerCase()) || 
        f.desc.toLowerCase().includes(filterText.toLowerCase())
      );

      filtered.forEach(file => {
        const prog = getProgress(file.id);
        const isUnlocked = prog.unlocked;
        const count = prog.count;
        const pct = Math.min(100, Math.round((count / file.requiredAds) * 100));

        const card = document.createElement('div');
        card.className = 'card';
        card.innerHTML = `
          <div class="card-top">
            <div class="thumb-wrap" onclick="showImage('${file.img}')">
              <img src="${file.img}" alt="${file.title}">

                <span>${isUnlocked ? '🔓 আনলক সম্পন্ন' : '🔒 লকড'}</span>
                <strong>${count}/${file.requiredAds}</strong>
              </div>
              <div class="progress-bar-bg">
                <div class="progress-bar-fill ${isUnlocked ? 'unlocked' : ''}" style="width: ${pct}%;"></div>
              </div>
            </div>
          </div>
          <div class="card-actions">
            ${isUnlocked ? `
              <button class="btn btn-ad unlocked">✅ আনলকড</button>
            ` : `
              <button class="btn btn-ad" onclick="watchRewardedAd('${file.id}')">✨ বিজ্ঞাপন দেখুন (+১)</button>
            `}
