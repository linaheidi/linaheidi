// Declare global Monetag SDK functions
declare global {
  interface Window {
    show_11999946?: (format?: string) => Promise<void>;
    show_9837009?: any;
    __miniApp?: any;
  }
}

interface Banner {
  id: string;
  title: string;
  desc: string;
  yt: string;
  img: string;
}

interface BotFile {
  id: string;
  title: string;
  desc: string;
  img: string;
  requiredAds: number;
  downloadUrl: string;
  yt: string;
}

interface ProgressState {
  count: number;
  unlocked: boolean;
}

const DEFAULT_IMG =
  'data:image/svg+xml;utf8,' +
  encodeURIComponent(
    `<svg xmlns='http://www.w3.org/2000/svg' width='640' height='360'><rect width='100%' height='100%' fill='#e6ecff'/><text x='50%' y='50%' dominant-baseline='middle' text-anchor='middle' fill='#6b6f85' font-size='22'>DEFAULT IMAGE</text></svg>`

export default function App() {
  // App data as provided in user request
  const banners: Banner[] = [
    {
      id: 'b1',
      title: 'EARNING APP',
      desc: 'Watch ads to unlock files - Banner 1',
      yt: 'https://youtube.com/@the-eduverse?si=mmhmqaUxv_odGWp0',
      img: 'https://trickbd.com/wp-content/uploads/2025/09/07/img-to-url-.png',
    },
    {
      id: 'b2',
      title: 'TELEGRAM MINI APP',
      desc: 'Create mini apps easily',

    },
    {
      id: 'b3',
      title: 'NO CODING REQUIRED',
      desc: 'Monetag + Telegram',
      yt: 'https://youtube.com/@the-eduverse?si=mmhmqaUxv_odGWp0',
      img: 'https://trickbd.com/wp-content/uploads/2025/09/07/img-to-url-.png',
    },
  ];

  const files: BotFile[] = [
    {
      id: 'file1',
      title: 'Telegram Earning Bot',
      desc: 'Download the files. Watch 10 Ads to Unlock',
      img: 'https://trickbd.com/wp-content/uploads/2025/09/07/20250907_161319.jpg',
      requiredAds: 10,
      downloadUrl: 'https://t.me/trickmimicrojob/19',
      yt: 'https://youtube.com/@the-eduverse?si=mmhmqaUxv_odGWp0',

    {

      desc: 'Download the files. Watch 20 Ads to Unlock',
      img: 'https://trickbd.com/wp-content/uploads/2025/09/07/20250907_170245.jpg',
      requiredAds: 20,
      downloadUrl: 'https://t.me/trickmimicrojob/19',
      yt: 'https://youtube.com/@the-eduverse?si=mmhmqaUxv_odGWp0',

      title: 'Earning Bot Source Code With Admin Panel',
      desc: '(File) Watch Ads to Unlock',
      img: 'https://trickbd.com/wp-content/uploads/2025/09/07/20250907_171832.jpg',
      requiredAds: 15,
      downloadUrl: 'https://t.me/trickmimicrojob/19',
      yt: 'https://youtube.com/@the-eduverse?si=mmhmqaUxv_odGWp0',
    },
  ];

  // States
  const [searchTerm, setSearchTerm] = useState('');
  const [currentSlide, setCurrentSlide] = useState(0);
  const [progressMap, setProgressMap] = useState<Record<string, ProgressState>>({});
  const [sdkAvailable, setSdkAvailable] = useState<boolean>(false);
  const [toastMessage, setToastMessage] = useState<{ text: string; type: 'success' | 'info' | 'warn' } | null>(null);

  // Modals state
  const [previewImage, setPreviewImage] = useState<string | null>(null);
  const [youtubeUrl, setYoutubeUrl] = useState<string | null>(null);

  // Simulated ad modal state (used when testing or when Monetag SDK is blocked by browser ad blocker)
  const [simulatedAd, setSimulatedAd] = useState<{
    fileId: string;
    fileTitle: string;
    timeLeft: number;
    reasonMessage: string;
  } | null>(null);

  // Helper to load progress from localStorage
  const getProgressKey = (id: string) => 'unlock_progress_' + id;

  const loadProgress = (id: string): ProgressState => {
    try {
      const raw = localStorage.getItem(getProgressKey(id));
      if (!raw) return { count: 0, unlocked: false };
      return JSON.parse(raw);
    } catch {
      return { count: 0, unlocked: false };
    }
  };

  const saveProgress = (id: string, state: ProgressState) => {
    try {
      localStorage.setItem(getProgressKey(id), JSON.stringify(state));
    } catch (e) {
      console.error('Failed to save to localStorage', e);
    }
    setProgressMap((prev) => ({ ...prev, [id]: state }));
    window.dispatchEvent(new CustomEvent('progressUpdated', { detail: { id, state } }));
  };

  // Toast notification helper
  const showToast = (text: string, type: 'success' | 'info' | 'warn' = 'info') => {
    setToastMessage({ text, type });
    setTimeout(() => {
      setToastMessage(null);
    }, 4000);
  };

  // Increment ad count and unlock if threshold is reached
  const incrementAdCount = (fileId: string, increment = 1, reason = 'manual') => {
    const file = files.find((f) => f.id === fileId);
    if (!file) return;

    const current = loadProgress(fileId);
    const newCount = (current.count || 0) + increment;
    const isNowUnlocked = newCount >= file.requiredAds;

    const updatedState: ProgressState = {
      count: isNowUnlocked ? file.requiredAds : newCount,
      unlocked: isNowUnlocked || current.unlocked,
    };

    saveProgress(fileId, updatedState);

    if (isNowUnlocked && !current.unlocked) {
      showToast(`🎉 অভিনন্দন! "${file.title}" ফাইলটি আনলক হয়েছে! এখন ডাউনলোড করতে পারেন।`, 'success');
    } else {
      showToast(`+${increment} বিজ্ঞাপন সফলভাবে যোগ হয়েছে (${updatedState.count}/${file.requiredAds}) [${reason}]`, 'info');
    }
  };

  // Initial load
  useEffect(() => {
    const initialMap: Record<string, ProgressState> = {};
    files.forEach((file) => {
      initialMap[file.id] = loadProgress(file.id);
    });
    setProgressMap(initialMap);

    // Check Monetag SDK availability
    const checkSdk = () => {
      if (typeof window.show_11999946 === 'function') {
        setSdkAvailable(true);
      } else {
        setSdkAvailable(false);
      }
    };

    checkSdk();
    const interval = setInterval(checkSdk, 1500);

    // Carousel auto-play
    const slideTimer = setInterval(() => {
      setCurrentSlide((prev) => (prev + 1) % banners.length);
    }, 4500);

    // Global listener for cross-window / event sync
    const handleProgressUpdate = (e: any) => {
      if (e.detail?.id && e.detail?.state) {
        setProgressMap((prev) => ({ ...prev, [e.detail.id]: e.detail.state }));
      }
    };
    window.addEventListener('progressUpdated', handleProgressUpdate);

    // Expose debug / legacy API for window console testing
    window.__miniApp = {
      data: { banners, files },
      loadProgress,
      saveProgress,
      incrementAdCount,
      showRewardedPopup: () => {
        if (typeof window.show_11999946 === 'function') {
          return window.show_11999946('pop');
        }
        return Promise.reject(new Error('SDK not loaded'));
      },
    };

    return () => {
      clearInterval(interval);
      clearInterval(slideTimer);
      window.removeEventListener('progressUpdated', handleProgressUpdate);
    };
  }, []);

  // Timer countdown for simulated ad modal
  useEffect(() => {
    if (!simulatedAd || simulatedAd.timeLeft <= 0) return;

    const timer = setInterval(() => {
      setSimulatedAd((prev) => {
        if (!prev) return null;
        if (prev.timeLeft <= 1) {
          return { ...prev, timeLeft: 0 };
        }
        return { ...prev, timeLeft: prev.timeLeft - 1 };
      });
    }, 1000);

    return () => clearInterval(timer);
  }, [simulatedAd?.timeLeft]);

  // Handle Rewarded Popup Trigger (The exact new integration requested by the user!)
  const handleWatchRewardedAd = (fileId: string) => {
    const file = files.find((f) => f.id === fileId);
    if (!file) return;

    // Check if the new Monetag SDK function show_11999946 is ready
    if (typeof window.show_11999946 === 'function') {
      showToast('বিজ্ঞাপন লোড হচ্ছে (Rewarded Popup)... অনুগ্রহ করে অপেক্ষা করুন।', 'info');

      // USER'S EXACT CODE INTEGRATION:
      // show_11999946('pop').then(() => { ... }).catch(e => { ... })
      window.show_11999946('pop')
        .then(() => {
          // User watched ad till the end or closed it in interstitial format
          // Your code to reward user for rewarded format:
          incrementAdCount(fileId, 1, 'Rewarded Pop 11999946');
        })
        .catch((e: any) => {
          // User got error during playing ad
          console.warn('Ad playback error or closed before reward:', e);
          // If ad failed (e.g. adblocker, network, or format unfulfilled), allow simulated fallback so user is not stuck:
          setSimulatedAd({
            fileId,
            fileTitle: file.title,
            timeLeft: 12,
            reasonMessage: 'বিজ্ঞাপন প্রদশনে সমস্যা হয়েছে বা ব্রাউজারে অ্যাড ব্লকার সক্রিয়। বিকল্প ডেমো সিস্টেমে প্রগ্রেস সম্পন্ন করুন:',
          });
        });
    } else if (typeof window.show_9837009 === 'function') {
      // Legacy fallback
      showToast('পূর্ববর্তী বিজ্ঞাপন জোন লোড হচ্ছে...', 'info');
      try {
        const res = window.show_9837009((event: any) => {
          incrementAdCount(fileId, 1, 'SDK Legacy');
        });
        if (res && typeof res.then === 'function') {
          res
            .then(() => incrementAdCount(fileId, 1, 'SDK Legacy Promise'))
            .catch(() => {
              setSimulatedAd({
                fileId,
                fileTitle: file.title,
                timeLeft: 10,
                reasonMessage: 'বিজ্ঞাপন বন্ধ হয়েছে। ডেমো ভিউ সম্পন্ন করুন:',
              });
            });
        }
      } catch {
        setSimulatedAd({
          fileId,
          fileTitle: file.title,
          timeLeft: 12,
          reasonMessage: 'অ্যাড নেটওয়ার্ক লোড হচ্ছে। নিচের ডেমো টাইমার শেষে আনলক করুন:',
        });
      }
    } else {
      // SDK not loaded yet or blocked by user's AdBlock extension
      setSimulatedAd({
        fileId,
        fileTitle: file.title,
        timeLeft: 12,
        reasonMessage: 'Monetag SDK (Zone: 11999946) লোড করা সম্ভব হয়নি (সম্ভবত অ্যাড-ব্লকার সক্রিয়)। টেস্টিং ও ডাউনলোডের জন্য নিচের ডেমো বিজ্ঞাপন টাইমার ব্যবহার করুন:',
      });
    }
  };

  // YouTube modal helper
  const openYouTubeModal = (url: string) => {
    setYoutubeUrl(url);
  };

  // Format YouTube Embed URL
  const getEmbedUrl = (ytUrl: string) => {
    try {
      const parsed = new URL(ytUrl);
      if (parsed.hostname.includes('youtube.com') || parsed.hostname.includes('youtu.be')) {
        const v = parsed.searchParams.get('v');
        if (v) return `https://www.youtube.com/embed/${v}`;
        if (parsed.pathname.includes('/embed/')) return ytUrl;
      }
    } catch {
      // fallback
    }
    return null;
  };

  // Filtered files based on search
  const filteredFiles = files.filter((f) => {
    if (!searchTerm.trim()) return true;
    const term = searchTerm.toLowerCase();
    return f.title.toLowerCase().includes(term) || f.desc.toLowerCase().includes(term);
  });

  // Reset all progress (testing helper)
  const resetAllProgress = () => {
    if (confirm('আপনি কি সমস্ত প্রগ্রেস রিসেট করতে চান?')) {
      files.forEach((f) => {
        localStorage.removeItem(getProgressKey(f.id));
      });
      const resetMap: Record<string, ProgressState> = {};
      files.forEach((f) => {
        resetMap[f.id] = { count: 0, unlocked: false };
      });
      setProgressMap(resetMap);
      showToast('সব প্রগ্রেস রিসেট করা হয়েছে।', 'info');
    }
  };

  return (
    <div className="min-h-screen bg-[#f5f7fb] text-slate-800 pb-16 font-sans">
      {/* Toast Notification */}
      {toastMessage && (
        <div className="fixed top-4 right-4 z-50 max-w-sm animate-fade-in transition-all">
          <div
            className={`flex items-start gap-3 p-4 rounded-xl shadow-xl border text-sm backdrop-blur-md ${
              toastMessage.type === 'success'
                ? 'bg-emerald-500/95 text-white border-emerald-400'
                : toastMessage.type === 'warn'
                ? 'bg-amber-500/95 text-white border-amber-400'
                : 'bg-indigo-600/95 text-white border-indigo-400'
            }`}
          >
            {toastMessage.type === 'success' ? (
              <CheckCircle2 className="w-5 h-5 flex-shrink-0 mt-0.5" />
            ) : (
              <Info className="w-5 h-5 flex-shrink-0 mt-0.5" />
            )}
            <div className="flex-1 font-medium">{toastMessage.text}</div>
            <button
              onClick={() => setToastMessage(null)}
              className="text-white/80 hover:text-white transition-colors"
            >
              <X className="w-4 h-4" />
            </button>
          </div>
        </div>
      )}

      {/* Main Container */}
      <div className="max-w-4xl mx-auto px-4 pt-4 sm:pt-6">
        {/* Header */}
        <header className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4 mb-5 pb-2">
          <div className="space-y-1">
            <div className="flex items-center gap-2">
              <h1 className="text-xl sm:text-2xl font-bold tracking-tight text-slate-900 flex items-center gap-1.5">
                Welcome <span className="animate-pulse">🥰</span>
              </h1>
              {sdkAvailable ? (
                <span className="inline-flex items-center gap-1 px-2.5 py-0.5 rounded-full text-xs font-semibold bg-emerald-100 text-emerald-800 border border-emerald-200">
                  <span className="w-1.5 h-1.5 rounded-full bg-emerald-500 animate-ping"></span>
                  Monetag SDK Ready (Zone 11999946)
                </span>
              ) : (
                <span className="inline-flex items-center gap-1 px-2.5 py-0.5 rounded-full text-xs font-medium bg-indigo-50 text-indigo-700 border border-indigo-100">
                  <Radio className="w-3 h-3 text-indigo-500" />
                  Zone: 11999946
                </span>
              )}
            </div>
            <p className="text-xs sm:text-sm text-slate-500">
              বিজ্ঞাপন দেখুন, প্রগ্রেস পূরণ করুন এবং ডাউনলোড আনলক করুন
            </p>
          </div>

          {/* Search Box */}
          <div className="w-full sm:w-72 relative">
            <input
              id="searchInput"
              type="text"
              value={searchTerm}
              onChange={(e) => setSearchTerm(e.target.value)}
              placeholder="ফাইলের জন্য অনুসন্ধান করুন..."
              className="w-full pl-10 pr-4 py-2.5 text-sm rounded-full bg-white border border-slate-200/90 shadow-sm focus:outline-none focus:ring-2 focus:ring-indigo-500/20 focus:border-indigo-500 transition-all placeholder:text-slate-400"
            />
            <Search className="w-4 h-4 text-slate-400 absolute left-3.5 top-1/2 -translate-y-1/2 pointer-events-none" />
            {searchTerm && (
              <button
                onClick={() => setSearchTerm('')}
                className="absolute right-3 top-1/2 -translate-y-1/2 text-slate-400 hover:text-slate-600"
              >
                <X className="w-3.5 h-3.5" />
              </button>
            )}
          </div>
        </header>

        {/* Carousel / Banner Section */}
        <div className="relative rounded-2xl overflow-hidden shadow-md bg-white border border-slate-100 mb-6 group">
          <div
            className="flex transition-transform duration-500 ease-out"
            style={{ transform: `translateX(-${currentSlide * 100}%)` }}
          >
            {banners.map((banner, index) => (
              <div
                key={banner.id}
                className="min-w-full p-4 sm:p-5 flex flex-col sm:flex-row items-center gap-4 sm:gap-6 bg-gradient-to-r from-white via-indigo-50/30 to-purple-50/20"
              >
                <div
                  onClick={() => setPreviewImage(banner.img || DEFAULT_IMG)}
                  className="relative cursor-pointer group/img flex-shrink-0 overflow-hidden rounded-xl border border-slate-200/80 shadow-sm"
                >
                  <img
                    src={banner.img || DEFAULT_IMG}
                    alt={banner.title}
                    className="w-28 sm:w-36 h-20 sm:h-24 object-cover transform transition-transform duration-300 group-hover/img:scale-105"
                    onError={(e) => {
                      (e.target as HTMLImageElement).src = DEFAULT_IMG;
                    }}
                  />
                  <div className="absolute inset-0 bg-black/20 opacity-0 group-hover/img:opacity-100 transition-opacity flex items-center justify-center text-white text-xs font-semibold">
                    বড় করে দেখুন
                  </div>
                </div>

                <div className="flex-1 text-center sm:text-left space-y-1.5">
                  <div className="inline-block px-2 py-0.5 text-[11px] font-bold uppercase tracking-wider text-indigo-700 bg-indigo-100/80 rounded-md">
                    Featured
                  </div>
                  <h3 className="text-base sm:text-lg font-bold text-slate-800 leading-snug">
                    {banner.title}
                  </h3>
                  <p className="text-xs sm:text-sm text-slate-500">{banner.desc}</p>
                  <div className="flex items-center justify-center sm:justify-start gap-2.5 pt-2">
                    <button
                      onClick={() => openYouTubeModal(banner.yt)}
                      className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-xs font-semibold text-white bg-gradient-to-r from-[#6b8cff] to-[#b44bff] shadow-sm hover:opacity-95 active:scale-95 transition-all"
                    >
                      <Youtube className="w-3.5 h-3.5 text-red-100" />
                      Watch Video
                    </button>
                    <button
                      onClick={() => setPreviewImage(banner.img || DEFAULT_IMG)}
                      className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-xs font-semibold text-slate-700 bg-white border border-slate-200 hover:bg-slate-50 active:scale-95 transition-all"
                    >
                      <ImageIcon className="w-3.5 h-3.5 text-slate-500" />
                      Preview
                    </button>
                  </div>
                </div>
              </div>
            ))}
          </div>

          {/* Dots */}
          <div className="absolute bottom-3 left-1/2 -translate-x-1/2 flex items-center gap-1.5 z-10 bg-black/10 backdrop-blur-xs px-2.5 py-1 rounded-full">
            {banners.map((_, idx) => (
              <button
                key={idx}
                onClick={() => setCurrentSlide(idx)}
                className={`transition-all rounded-full ${
                  currentSlide === idx
                    ? 'w-5 h-2 bg-indigo-600'
                    : 'w-2 h-2 bg-slate-300 hover:bg-slate-400'
                }`}
                aria-label={`Go to slide ${idx + 1}`}
              />
            ))}
          </div>

          {/* Slide Arrow controls */}
          <button
            onClick={() =>
              setCurrentSlide((prev) => (prev - 1 + banners.length) % banners.length)
            }
            className="absolute left-2 top-1/2 -translate-y-1/2 w-7 h-7 rounded-full bg-white/80 hover:bg-white shadow flex items-center justify-center text-slate-700 opacity-0 group-hover:opacity-100 transition-opacity"
            aria-label="Previous slide"
          >
            <ChevronLeft className="w-4 h-4" />
          </button>
          <button
            onClick={() => setCurrentSlide((prev) => (prev + 1) % banners.length)}
            className="absolute right-2 top-1/2 -translate-y-1/2 w-7 h-7 rounded-full bg-white/80 hover:bg-white shadow flex items-center justify-center text-slate-700 opacity-0 group-hover:opacity-100 transition-opacity"
            aria-label="Next slide"
          >
            <ChevronRight className="w-4 h-4" />
          </button>
        </div>

        {/* Section Header */}
        <div className="flex items-center justify-between mb-4">
          <div className="flex items-center gap-2">
            <h2 className="text-lg sm:text-xl font-bold text-slate-900 tracking-tight">
              উপলব্ধ ফাইল
            </h2>
            <span className="px-2 py-0.5 text-xs font-semibold bg-indigo-100 text-indigo-700 rounded-full">
              {filteredFiles.length} টি আইটেম
            </span>
          </div>

          <button
            onClick={resetAllProgress}
            className="text-xs text-slate-400 hover:text-slate-600 flex items-center gap-1 transition-colors"
            title="প্রগ্রেস রিসেট করুন"
          >
            <RotateCcw className="w-3 h-3" />
            রিসেট
          </button>
        </div>

        {/* Cards Grid */}
        <div className="grid grid-cols-1 md:grid-cols-2 gap-4 mb-10">
          {filteredFiles.length === 0 ? (
            <div className="col-span-full py-12 text-center bg-white rounded-2xl border border-dashed border-slate-300">
              <Search className="w-8 h-8 text-slate-300 mx-auto mb-2" />
              <p className="text-slate-500 font-medium text-sm">কোন ফাইল পাওয়া যায়নি</p>
              <button
                onClick={() => setSearchTerm('')}
                className="mt-3 text-xs text-indigo-600 font-semibold underline hover:text-indigo-800"
              >
                সব ফাইল দেখুন
              </button>
            </div>
          ) : (
            filteredFiles.map((file) => {
              const state = progressMap[file.id] || { count: 0, unlocked: false };
              const isUnlocked = !!state.unlocked;
              const count = state.count || 0;
              const percent = Math.min(
                100,
                file.requiredAds > 0 ? Math.round((count / file.requiredAds) * 100) : 100
              );

              return (
